cmake_minimum_required(VERSION 3.20)
project(field_ngx LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Vulkan REQUIRED)
if(NOT DEFINED DLSS_ROOT)
  message(FATAL_ERROR "Pass -DDLSS_ROOT=<pinned NVIDIA/DLSS checkout>; this optional build never downloads SDKs")
endif()
add_library(field_ngx SHARED field_ngx.cpp)
target_include_directories(field_ngx SYSTEM PRIVATE "${DLSS_ROOT}/include")
target_link_libraries(field_ngx PRIVATE "${DLSS_ROOT}/lib/Linux_x86_64/libnvsdk_ngx.a" Vulkan::Vulkan dl pthread)
target_compile_options(field_ngx PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)

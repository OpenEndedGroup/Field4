cmake_minimum_required(VERSION 3.24)
project(field_loris LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LORIS_PYTHON OFF CACHE BOOL "" FORCE)
set(LORIS_UTILS OFF CACHE BOOL "" FORCE)
set(LORIS_FFTW OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${FIELD_LORIS_SOURCE}/src/Analyzer.h")
  message(FATAL_ERROR "Set FIELD_LORIS_SOURCE to the pinned Loris source tree")
endif()
add_subdirectory("${FIELD_LORIS_SOURCE}" loris)
set_target_properties(loris PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(field_loris main.cpp)
target_link_libraries(field_loris PRIVATE loris)
set_target_properties(field_loris PROPERTIES BUILD_RPATH "$ORIGIN/loris/src;@loader_path/loris/src")
if(WIN32)
  add_custom_command(TARGET field_loris POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:loris> $<TARGET_FILE_DIR:field_loris>)
endif()
